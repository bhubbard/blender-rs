blender-rs

A memory-safe, SIMD-accelerated Rust re-architecture of the Blender 3D creation suite (blenlib, makesdna, bmesh, blenkernel, and io). Replaces raw cyclic C pointers with generational arena handles (MemPool<T>) and pointerless spatial trees (Bvh, KdTree).

RUST SDK // HALF-EDGE BMESH & BVH RAYCAST 
use blender_bmesh::BMesh;
use blender_geom::bvh::Bvh;
use blender_math::Vec3;

fn main() -> anyhow::Result<()> {
    // 1. Construct a quad in the generational BMesh kernel
    let mut bm = BMesh::new();
    let v0 = bm.vert_create(Some(Vec3::new(0.0, 0.0, 0.0)), None);
    let v1 = bm.vert_create(Some(Vec3::new(2.0, 0.0, 0.0)), None);
    let v2 = bm.vert_create(Some(Vec3::new(2.0, 2.0, 0.0)), None);
    let v3 = bm.vert_create(Some(Vec3::new(0.0, 2.0, 0.0)), None);
    let face = bm.face_create(&[v0, v1, v2, v3], None)?;

    // 2. Split an edge via Euler operator (promotes quad to 5-gon)
    let first_edge = bm.epool.iter().next().unwrap().0;
    bm.edge_split(first_edge, 0.5)?;
    bm.recalc_face_normals();
    assert_eq!(bm.fpool.get(face).unwrap().len, 5);

    // 3. Raycast triangles with blender_geom::bvh::Bvh
    let tri = [Vec3::ZERO, Vec3::X * 2.0, Vec3::Y * 2.0];
    let bvh = Bvh::new(&[tri]);
    let hit = bvh.ray_cast(Vec3::new(0.5, 0.5, 5.0), -Vec3::Z).unwrap();
    assert!((hit.t - 5.0).abs() < 1e-6);
    Ok(())
}
